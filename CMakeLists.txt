cmake_minimum_required(VERSION 3.25)
project(Vellum VERSION 0.1.0 LANGUAGES CXX)
option(VELLUM_OFFICIAL_RELEASE "Build a signed official release" OFF)
set(VELLUM_SIGN_CERT_SHA1 "" CACHE STRING "SHA-1 thumbprint of the Authenticode certificate")
set(VELLUM_TIMESTAMP_URL "http://timestamp.digicert.com" CACHE STRING "RFC 3161 timestamp service")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Vellum requires Windows x64.")
endif()
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Widgets Test)
get_target_property(VELLUM_QMAKE Qt6::qmake IMPORTED_LOCATION)
get_filename_component(VELLUM_QT_BIN "${VELLUM_QMAKE}" DIRECTORY)
execute_process(COMMAND "${VELLUM_QMAKE}" -query QT_INSTALL_PLUGINS
    OUTPUT_VARIABLE VELLUM_QT_PLUGINS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(SODIUM_ROOT "${CMAKE_SOURCE_DIR}/.deps/sodium/libsodium" CACHE PATH "Pinned libsodium MSVC package")
if(NOT Qt6_VERSION VERSION_EQUAL "6.10.3")
    message(FATAL_ERROR "Vellum requires the audited Qt 6.10.3 dependency set.")
endif()
include(cmake/DependencyHashes.cmake)
add_library(sodium SHARED IMPORTED)
set_target_properties(sodium PROPERTIES IMPORTED_IMPLIB "${SODIUM_ROOT}/x64/Release/v143/dynamic/libsodium.lib" IMPORTED_LOCATION "${SODIUM_ROOT}/x64/Release/v143/dynamic/libsodium.dll" INTERFACE_INCLUDE_DIRECTORIES "${SODIUM_ROOT}/include")
add_library(VaultCore src/VaultStore.cpp src/VaultStore.h src/RecoveryCopy.cpp src/RecoveryCopy.h src/PassphrasePolicy.cpp src/PassphrasePolicy.h)
target_link_libraries(VaultCore PUBLIC Qt6::Core sodium Advapi32)
target_include_directories(VaultCore PUBLIC src)
qt_add_executable(Vellum WIN32 src/main.cpp src/VaultWindow.cpp src/VaultWindow.h src/SecretClipboard.cpp src/SecretClipboard.h)
target_link_libraries(Vellum PRIVATE VaultCore Qt6::Widgets Wtsapi32)
target_compile_definitions(Vellum PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(VaultCore PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(Vellum PRIVATE /W4 /permissive- /utf-8)
endif()
enable_testing()
qt_add_executable(VaultTests tests/VaultTests.cpp)
target_link_libraries(VaultTests PRIVATE VaultCore Qt6::Test)
add_test(NAME VaultTests COMMAND VaultTests)
add_custom_command(TARGET VaultTests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:sodium>" "$<TARGET_FILE_DIR:VaultTests>")
install(TARGETS Vellum RUNTIME DESTINATION .)
install(FILES "$<TARGET_FILE:sodium>" DESTINATION .)
qt_add_executable(VaultUiTests tests/VaultUiTests.cpp src/VaultWindow.cpp src/VaultWindow.h src/SecretClipboard.cpp src/SecretClipboard.h)
target_link_libraries(VaultUiTests PRIVATE VaultCore Qt6::Widgets Qt6::Test Wtsapi32)
target_compile_definitions(VaultUiTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(VaultTests VaultUiTests PROPERTIES WIN32_EXECUTABLE FALSE)
if(MSVC)
    target_compile_options(VaultUiTests PRIVATE /utf-8)
endif()
if(VELLUM_OFFICIAL_RELEASE)
    if(NOT VELLUM_SIGN_CERT_SHA1 MATCHES "^[0-9A-Fa-f]{40}$")
        message(FATAL_ERROR "Official releases require VELLUM_SIGN_CERT_SHA1 with a 40-character certificate thumbprint.")
    endif()
    find_program(VELLUM_SIGNTOOL signtool REQUIRED)
    add_custom_command(TARGET Vellum POST_BUILD
        COMMAND "${VELLUM_SIGNTOOL}" sign /sha1 "${VELLUM_SIGN_CERT_SHA1}" /fd SHA256
            /tr "${VELLUM_TIMESTAMP_URL}" /td SHA256 "$<TARGET_FILE:Vellum>"
        COMMAND "${VELLUM_SIGNTOOL}" verify /pa /all "$<TARGET_FILE:Vellum>"
        COMMENT "Signing and verifying Vellum.exe"
        VERBATIM)
endif()
add_test(NAME VaultUiTests COMMAND VaultUiTests)
set_tests_properties(VaultUiTests PROPERTIES ENVIRONMENT
    "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${VELLUM_QT_PLUGINS}/platforms")

# Ship only the runtime modules and platform plugins used by this application.
find_program(VELLUM_WINDEPLOYQT windeployqt HINTS "${VELLUM_QT_BIN}" REQUIRED NO_DEFAULT_PATH)
configure_file(cmake/DeployRuntime.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/DeployRuntime.cmake" @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployRuntime.cmake")
install(FILES README.md THIRD-PARTY-NOTICES.txt DESTINATION .)
install(DIRECTORY licenses/ DESTINATION licenses)
install(DIRECTORY docs/ DESTINATION docs)

set(CPACK_PACKAGE_NAME "Vellum")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
if(VELLUM_OFFICIAL_RELEASE)
    set(CPACK_PACKAGE_FILE_NAME "Vellum-${PROJECT_VERSION}-Windows-x64")
else()
    set(CPACK_PACKAGE_FILE_NAME "Vellum-${PROJECT_VERSION}-Windows-x64-UNSIGNED-DEVELOPMENT")
endif()
set(CPACK_GENERATOR ZIP)
include(CPack)
